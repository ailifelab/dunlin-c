cmake_minimum_required(VERSION 3.20)
project(dunlin_agent_scheduler LANGUAGES C)
find_package(SQLite3 3.34 REQUIRED)
add_library(dunlin_agent_scheduler STATIC scheduler.c)
target_include_directories(dunlin_agent_scheduler PUBLIC ../../include)
target_compile_features(dunlin_agent_scheduler PUBLIC c_std_11)
target_link_libraries(dunlin_agent_scheduler PUBLIC SQLite::SQLite3)
include(CTest)
if(BUILD_TESTING)
 add_executable(scheduler_test ../../tests/agent_scheduler.c)
 target_link_libraries(scheduler_test PRIVATE dunlin_agent_scheduler)
 if(MSVC)
  target_compile_options(scheduler_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(scheduler_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME dunlin.agent-scheduler COMMAND scheduler_test)
endif()
