cmake_minimum_required(VERSION 3.20)
project(dunlin_agent_mailbox LANGUAGES C)
include(CTest)
find_package(SQLite3 3.34 REQUIRED)
get_filename_component(DUNLIN_MAILBOX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(dunlin_agent_mailbox STATIC mailbox.c "${DUNLIN_MAILBOX_ROOT}/modules/content/hash.c")
add_library(DunlinC::AgentMailbox ALIAS dunlin_agent_mailbox)
target_include_directories(dunlin_agent_mailbox PUBLIC "${DUNLIN_MAILBOX_ROOT}/include")
target_compile_features(dunlin_agent_mailbox PUBLIC c_std_11)
target_link_libraries(dunlin_agent_mailbox PUBLIC SQLite::SQLite3)
set_target_properties(dunlin_agent_mailbox PROPERTIES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(dunlin_agent_mailbox PRIVATE /utf-8 /W4)
else()
    target_compile_options(dunlin_agent_mailbox PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(dunlin_agent_mailbox_test "${DUNLIN_MAILBOX_ROOT}/tests/agent_mailbox.c")
    target_link_libraries(dunlin_agent_mailbox_test PRIVATE DunlinC::AgentMailbox)
    if(MSVC)
        target_compile_options(dunlin_agent_mailbox_test PRIVATE /utf-8 /UNDEBUG)
    else()
        target_compile_options(dunlin_agent_mailbox_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME dunlin.agent-mailbox.standalone COMMAND dunlin_agent_mailbox_test)
endif()
