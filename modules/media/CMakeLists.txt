cmake_minimum_required(VERSION 3.20)
project(dunlin_media LANGUAGES C)
add_library(dunlin_media STATIC session.c recording.c)
target_include_directories(dunlin_media PUBLIC ../../include)
target_compile_features(dunlin_media PUBLIC c_std_11)
include(CTest)
if(BUILD_TESTING)
 add_executable(media_test ../../tests/media_session.c)
 target_link_libraries(media_test PRIVATE dunlin_media)
 if(MSVC)
  target_compile_options(media_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(media_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME dunlin.media-session COMMAND media_test)
 add_executable(recording_test ../../tests/media_recording.c)
 target_link_libraries(recording_test PRIVATE dunlin_media)
 if(MSVC)
  target_compile_options(recording_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(recording_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME dunlin.media-recording COMMAND recording_test)
endif()
option(DUNLIN_MEDIA_REALTIME "Build optional libcurl WebSocket/Realtime transport" OFF)
if(DUNLIN_MEDIA_REALTIME)
 find_package(CURL 8.14 REQUIRED)
 find_package(OpenSSL REQUIRED)
 find_package(json-c CONFIG REQUIRED)
 add_library(dunlin_realtime STATIC websocket.c realtime.c ../../dunlin-compact/runtime/json.c)
 target_include_directories(dunlin_realtime PUBLIC ../../include)
 target_compile_features(dunlin_realtime PUBLIC c_std_11)
 target_link_libraries(dunlin_realtime PUBLIC CURL::libcurl OpenSSL::Crypto json-c::json-c)
 if(WIN32)
  target_link_libraries(dunlin_realtime PUBLIC ws2_32)
 endif()
endif()
